cmake_minimum_required(VERSION 3.22.1)
project(latentjam_equalizer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(lj_equalizer SHARED lj_equalizer.cpp)
target_compile_options(lj_equalizer PRIVATE -O3 -ffp-contract=off -fno-fast-math)
if(ANDROID)
    target_sources(lj_equalizer PRIVATE lj_equalizer_jni.cpp)
    # Keep stripped build IDs independent of the absolute checkout/build directories.
    target_compile_options(lj_equalizer PRIVATE
        "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=."
        "-fdebug-prefix-map=${CMAKE_BINARY_DIR}=.")
    target_link_options(lj_equalizer PRIVATE -Wl,-z,max-page-size=16384)
elseif(LJ_EQ_JNI)
    find_package(JNI REQUIRED)
    target_include_directories(lj_equalizer PRIVATE ${JNI_INCLUDE_DIRS})
    target_sources(lj_equalizer PRIVATE lj_equalizer_jni.cpp)
endif()
if(LJ_EQ_TESTS)
    enable_testing()
    add_executable(lj_equalizer_test tests/lj_equalizer_test.cpp)
    find_package(Threads REQUIRED)
    target_link_libraries(lj_equalizer_test PRIVATE lj_equalizer Threads::Threads)
    target_include_directories(lj_equalizer_test PRIVATE .)
    target_compile_options(lj_equalizer_test PRIVATE -ffp-contract=off -fno-fast-math)
    add_test(NAME equalizer COMMAND lj_equalizer_test)
endif()

if(APPLE AND LJ_EQ_TESTS)
    enable_language(OBJCXX)
    add_executable(lj_ios_effect_test tests/ios_effect_test.mm lj_equalizer_ios.mm)
    target_include_directories(lj_ios_effect_test PRIVATE .)
    target_compile_options(lj_ios_effect_test PRIVATE -fobjc-arc -ffp-contract=off -fno-fast-math)
    target_link_libraries(lj_ios_effect_test PRIVATE lj_equalizer "-framework AVFoundation" "-framework AudioToolbox" "-framework Foundation" objc)
    target_link_options(lj_ios_effect_test PRIVATE "-Wl,-rpath,$<TARGET_FILE_DIR:lj_equalizer>")
    add_test(NAME apple_audio_unit COMMAND lj_ios_effect_test)
endif()
